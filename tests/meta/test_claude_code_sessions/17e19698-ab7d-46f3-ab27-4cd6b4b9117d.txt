import os, subprocess, sys
from hypothesis import given, settings, strategies as st, HealthCheck

wt = sys.argv[1]
files = sorted(subprocess.run(["git", "-C", wt, "ls-files"], capture_output=True, text=True, check=True).stdout.split("\n"))
files = [f for f in files if f]
text = st.text(alphabet=st.characters(min_codepoint=32, max_codepoint=126), min_size=1, max_size=40)
done = []

@settings(database=None, max_examples=1, deadline=None, suppress_health_check=list(HealthCheck))
@given(st.data())
def mutate(data):
    if done:
        return
    chosen = data.draw(st.lists(st.sampled_from(files), min_size=5, max_size=15, unique=True))
    for rel in chosen:
        path = os.path.join(wt, rel)
        action = data.draw(st.sampled_from(["delete", "append", "replace"]))
        if action == "delete":
            os.remove(path)
            print(f"deleted  {rel}")
            continue
        with open(path, "rb") as fh:
            raw = fh.read()
        try:
            content = raw.decode("utf-8")
        except UnicodeDecodeError:
            action = "append"
            content = None
        new = data.draw(text)
        if action == "append" or not content or not content.splitlines():
            with open(path, "ab") as fh:
                fh.write(("\n" + new + "\n").encode())
            print(f"appended {rel}: {new!r}")
        else:
            lines = content.splitlines(keepends=True)
            i = data.draw(st.integers(0, len(lines) - 1))
            lines[i] = new + "\n"
            with open(path, "w", encoding="utf-8", newline="") as fh:
                fh.write("".join(lines))
            print(f"replaced {rel} line {i + 1}: {new!r}")
    done.append(True)

mutate()
print(f"{len(files)} tracked files, changed {len(done) and 'ok'}")
EOF
