import os, subprocess
from hypothesis import given, settings, strategies as st, HealthCheck

wt = os.environ["WT"]
files = subprocess.run(["git", "-C", wt, "ls-files"], capture_output=True, text=True, check=True).stdout.splitlines()
done = []

@settings(database=None, max_examples=1, suppress_health_check=list(HealthCheck), deadline=None)
@given(st.data())
def mutate(data):
    if done:
        return
    chosen = data.draw(st.lists(st.sampled_from(files), min_size=5, max_size=15, unique=True))
    text = st.text(alphabet="abcdefghijklmnopqrstuvwxyz ", min_size=1, max_size=40)
    for f in chosen:
        p = os.path.join(wt, f)
        action = data.draw(st.sampled_from(["delete", "append", "replace"]))
        if action == "delete":
            os.remove(p)
            print(f"deleted  {f}")
            continue
        with open(p, "rb") as fh:
            raw = fh.read()
        try:
            content = raw.decode("utf-8")
        except UnicodeDecodeError:
            os.remove(p)
            print(f"deleted  {f} (binary)")
            continue
        new = data.draw(text)
        if action == "append" or not content.splitlines():
            content += ("" if content.endswith("\n") or not content else "\n") + new + "\n"
            print(f"appended {f}: {new!r}")
        else:
            lines = content.splitlines(keepends=True)
            i = data.draw(st.integers(0, len(lines) - 1))
            lines[i] = new + "\n"
            content = "".join(lines)
            print(f"replaced {f} line {i + 1}: {new!r}")
        with open(p, "w", encoding="utf-8", newline="") as fh:
            fh.write(content)
    done.append(True)

mutate()
EOF
